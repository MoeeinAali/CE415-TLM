\begin{itemize}
    \item[(ب)] \textbf{مسئله‌ی پوشش راسی:} با فرض داشتن گراف $G$ و عدد صحیح $k$، آیا زیرمجموعه‌ی $S$ شامل $k$ راس از $G$ وجود دارد، به طوری که $S$ یک پوشش راسی $G$ باشد؟ (مجموعه‌ی $S$ یک پوشش راسی برای گراف $G$ است اگر و تنها اگر هر یال $G$ دارای حداقل یک راس داخل مجموعه‌ $S$ باشد.)

    \item[(ج)] \textbf{مسئله‌ی مجموعه‌ی غالب راسی:} با فرض داشتن گراف $G$ و عدد صحیح $k$، آیا زیرمجموعه‌ی $S$ شامل $k$ راس از $G$ وجود دارد، به طوری که $S$ یک مجموعه‌ی غالب $G$ باشد؟ (مجموعه‌ی $S$ یک مجموعه‌ی غالب برای گراف $G$ است اگر و تنها اگر هر راس $G$ یا داخل $S$ باشد یا با یک راس داخل $S$ همسایه باشد.)

    \item[(ه)] \textbf{مسئله LPATH:}
    $$ \text{LPATH} = \{ \langle G, a, b, k \rangle \mid G \text{ شامل یک مسیر ساده از } a \text{ به } b \text{ با طول حداقل } k \text{ باشد } \} $$
\end{itemize}
