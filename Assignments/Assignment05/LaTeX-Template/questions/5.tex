\subsectionaddtolist{ب}

ماشین تورینگ غیرقطعی $N$ را روی ورودی $\langle G, k \rangle$ به صورت زیر تعریف می‌کنیم:
\begin{enumerate}
    \item به صورت غیرقطعی یک زیرمجموعه $S \subseteq V$ با اندازه $k$ حدس بزن.
    \item برای هر یال $e = (u, v) \in E$ بررسی کن که حداقل یکی از دو سر آن در $S$ باشد.
    \item اگر همه یال‌ها پوشش داده شده بودند، {بپذیر}؛ در غیر این صورت {رد کن}.
\end{enumerate}
حدس زدن زیرمجموعه و بررسی یال‌ها در زمان چندجمله‌ای انجام می‌شود؛ بنابراین مسئله در NP است.

مسئله‌ی \lr{Independent Set} را به پوشش راسی کاهش می‌دهیم.

\begin{itemize}
    \item {روش کاهش:} فرض کنید یک گراف $G=(V, E)$ و عدد $m$ به عنوان ورودی مجموعه مستقل داده شده است. ورودی پوشش راسی را این‌گونه می‌سازیم: $G' = G$ و $k = |V| - m$. این کار در زمان چندجمله‌ای انجام می‌شود.

    \item {اثبات درستی:} باید نشان دهیم $G$ مجموعه مستقلی به اندازه $m$ دارد اگر و تنها اگر $G'$ پوشش راسی به اندازه $k = |V| - m$ داشته باشد.
    \begin{itemize}
        \item اگر $M$ یک مجموعه مستقل به اندازه $m$ در $G$ باشد، هیچ دو راسی در $M$ با هم یال ندارند. پس برای هر یال، حداقل یک سر آن در $V \setminus M$ است. بنابراین $V \setminus M$ یک پوشش راسی به اندازه $|V| - m$ است.
        \item اگر $C$ یک پوشش راسی به اندازه $|V|-m$ باشد، هیچ یالی وجود ندارد که هر دو سر آن در $V \setminus C$ باشد. پس $V \setminus C$ یک مجموعه مستقل به اندازه $m$ است.
    \end{itemize}
\end{itemize}

\subsectionaddtolist{ج}

ماشین تورینگ غیرقطعی $N$ را روی ورودی $\langle G, k \rangle$ به صورت زیر تعریف می‌کنیم:
\begin{enumerate}
    \item به صورت غیرقطعی یک زیرمجموعه $S \subseteq V$ با اندازه $k$ حدس بزن.
    \item برای هر راس $v \in V$ بررسی کن که یا $v \in S$ باشد یا حداقل با یک راس داخل $S$ همسایه باشد.
    \item اگر شرط برای همه رئوس برقرار بود، {بپذیر}؛ در غیر این صورت {رد کن}.
\end{enumerate}
این کار در زمان چندجمله‌ای انجام می‌شود؛ بنابراین مسئله در NP است.

مسئله‌ی پوشش راسی را به مجموعه غالب کاهش می‌دهیم.

\begin{itemize}
    \item {روش کاهش:} فرض کنید $(G=(V,E), k)$ ورودی پوشش راسی باشد (فرض می‌کنیم $G$ راس ایزوله ندارد). گراف $G'=(V',E')$ را این‌گونه می‌سازیم: برای هر یال $e=(u,v) \in E$، یک راس جدید $w_e$ می‌سازیم و آن را به $u$ و $v$ متصل می‌کنیم. پس $V' = V \cup \{w_e \mid e \in E\}$ و $E'$ شامل یال‌های $G$ به‌علاوه یال‌های جدید است. پارامتر را $k' = k$ می‌گیریم. ساخت این گراف در زمان چندجمله‌ای ممکن است.

    \item {اثبات درستی:}
    \begin{itemize}
        \item اگر $C$ یک پوشش راسی به اندازه $k$ در $G$ باشد، $C$ در $G'$ نیز مجموعه غالب است. هر راس اصلی $v \in V$ (چون ایزوله نیست) یا در $C$ است یا با آن همسایه است. هر راس جدید $w_e$ هم به دو سر یال $e$ متصل است و چون $C$ حداقل یکی از دو سر را دارد، $w_e$ نیز غالب می‌شود.
        \item اگر $D$ یک مجموعه غالب به اندازه $k$ در $G'$ باشد، ممکن است شامل برخی $w_e$ها باشد. اگر $w_e \in D$، آن را حذف کرده و به‌جای آن یکی از همسایه‌های اصلیش را در مجموعه قرار می‌دهیم. با تکرار این کار، به مجموعه غالب $D'$ به اندازه حداکثر $k$ می‌رسیم که فقط رئوس اصلی دارد. چون $D'$ باید همه $w_e$ها را غالب کند، برای هر یال $e$ حداقل یکی از دو سر آن در $D'$ است؛ پس $D'$ یک پوشش راسی برای $G$ است.
    \end{itemize}
\end{itemize}

\subsectionaddtolist{ه}

ماشین تورینگ غیرقطعی $N$ را روی ورودی $\langle G, a, b, k \rangle$ به صورت زیر تعریف می‌کنیم:
\begin{enumerate}
    \item به صورت غیرقطعی یک دنباله از رئوس $v_1, v_2, \dots, v_m$ حدس بزن.
    \item بررسی کن که $v_1 = a$ و $v_m = b$ باشد، همه رئوس متمایز باشند، بین هر دو راس متوالی یال وجود داشته باشد، و طول مسیر (یعنی $m-1$) حداقل $k$ باشد.
    \item اگر همه شرط‌ها برقرار بودند، {بپذیر}؛ در غیر این صورت {رد کن}.
\end{enumerate}
این بررسی در زمان چندجمله‌ای انجام می‌شود؛ بنابراین مسئله در NP است.

مسئله‌ی مسیر هامیلتونی بین دو راس مشخص را به \lr{LPATH} کاهش می‌دهیم.

\begin{itemize}
    \item {روش کاهش:} فرض کنید گراف $G=(V, E)$ و دو راس $s$ و $t$ ورودی مسیر هامیلتونی باشند. ورودی \lr{LPATH} را این‌گونه می‌سازیم: $G' = G$، $a = s$، $b = t$ و $k = |V| - 1$.

    \item {اثبات درستی:}
    \begin{itemize}
        \item اگر $G$ مسیر هامیلتونی از $s$ به $t$ داشته باشد، این مسیر از تمام $|V|$ راس دقیقاً یک بار عبور می‌کند؛ پس طول آن دقیقاً $|V| - 1$ است و شرط \lr{LPATH} برقرار می‌شود.
        \item اگر مسیر ساده‌ای از $a$ به $b$ با طول حداقل $|V| - 1$ وجود داشته باشد، چون طول مسیر ساده حداکثر $|V| - 1$ است، طول دقیقاً $|V| - 1$ خواهد بود. پس این مسیر از تمام رئوس عبور می‌کند و یک مسیر هامیلتونی از $s$ به $t$ است.
    \end{itemize}
\end{itemize}
