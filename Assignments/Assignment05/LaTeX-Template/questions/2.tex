
فرض کنید $L_1$ و $L_2$ دو زبان دلخواه در کلاس مربوطه باشند.

\subsection*{(آ) بسته بودن تحت اجتماع ($L_1 \cup L_2$)}

\paragraph{کلاس $\mathcal{P}$:}
چون $L_1, L_2 \in \mathcal{P}$، ماشین‌های تورینگ قطعی (DTM) مانند $M_1$ و $M_2$ وجود دارند که به ترتیب $L_1$ و $L_2$ را در زمان چندجمله‌ای $O(p_1(n))$ و $O(p_2(n))$ تصمیم‌گیری می‌کنند.
ماشین قطعی $M$ را برای $L_1 \cup L_2$ روی ورودی $x$ با طول $n$ به صورت زیر می‌سازیم:
\begin{enumerate}
    \item ماشین $M_1(x)$ را اجرا کن. اگر پذیرفت، \textbf{بپذیر}.
    \item ماشین $M_2(x)$ را اجرا کن. اگر پذیرفت، \textbf{بپذیر}؛ در غیر این صورت \textbf{رد کن}.
\end{enumerate}
زمان اجرای کل برابر $O(p_1(n) + p_2(n))$ است که چندجمله‌ای می‌باشد. بنابراین $L_1 \cup L_2 \in \mathcal{P}$.

\paragraph{کلاس $\mathcal{NP}$:}
چون $L_1, L_2 \in \mathcal{NP}$، ماشین‌های تورینگ غیرقطعی (NDTM) مانند $N_1$ و $N_2$ وجود دارند که در زمان چندجمله‌ای کار می‌کنند.
ماشین غیرقطعی $N$ را روی ورودی $x$ به صورت زیر تعریف می‌کنیم:
\begin{enumerate}
    \item به صورت غیرقطعی یکی از دو مسیر زیر را انتخاب کن:
    \begin{itemize}
        \item $N_1(x)$ را اجرا کن.
        \item $N_2(x)$ را اجرا کن.
    \end{itemize}
    \item اگر ماشین انتخاب‌شده ورودی را پذیرفت، \textbf{بپذیر}.
\end{enumerate}
زمان اجرا حداکثر $\max(\text{poly}_1(n), \text{poly}_2(n))$ است، پس $L_1 \cup L_2 \in \mathcal{NP}$.

%--------------------------------------------------
\subsection*{(ب) بسته بودن تحت اشتراک ($L_1 \cap L_2$)}

\paragraph{کلاس $\mathcal{P}$:}
ماشین قطعی $M$ برای $L_1 \cap L_2$ روی ورودی $x$:
\begin{enumerate}
    \item ماشین $M_1(x)$ را اجرا کن. اگر رد کرد، \textbf{رد کن}.
    \item ماشین $M_2(x)$ را اجرا کن. اگر پذیرفت، \textbf{بپذیر}؛ در غیر این صورت \textbf{رد کن}.
\end{enumerate}
زمان کل برابر $O(p_1(n) + p_2(n))$ و چندجمله‌ای است. بنابراین $L_1 \cap L_2 \in \mathcal{P}$.

\paragraph{کلاس $\mathcal{NP}$:}
با استفاده از مدل گواهی‌دهنده (Certificate): ورودی $x \in L_1 \cap L_2$ است اگر و تنها اگر گواهی $c_1$ برای $L_1$ و گواهی $c_2$ برای $L_2$ وجود داشته باشد.
گواهی جدید را $c = (c_1, c_2)$ در نظر می‌گیریم. ارزیاب (Verifier) در زمان چندجمله‌ای بررسی می‌کند که آیا هم $V_1(x, c_1)$ و هم $V_2(x, c_2)$ ورودی را می‌پذیرند یا خیر. زمان بررسی مجموع زمان‌های چندجمله‌ای است، پس $L_1 \cap L_2 \in \mathcal{NP}$.

%--------------------------------------------------
\subsection*{(ج) بسته بودن تحت الحاق ($L_1 \cdot L_2$)}

\paragraph{کلاس $\mathcal{P}$:}
برای ورودی $x$ به طول $n$:
\begin{enumerate}
    \item به ازای هر $i$ از $0$ تا $n$:
    \begin{itemize}
        \item رشته $x$ را به دو بخش $y = x[1..i]$ و $z = x[i+1..n]$ تقسیم کن.
        \item $M_1(y)$ و $M_2(z)$ را اجرا کن.
        \item اگر هر دو پذیرفتند، \textbf{بپذیر}.
    \end{itemize}
    \item اگر برای هیچ $i$ ای پذیرفته نشد، \textbf{رد کن}.
\end{enumerate}
تعداد تقسیم‌بندی‌ها $n+1$ حالت است و هر تست در زمان چندجمله‌ای انجام می‌شود. زمان کل $O(n \cdot (\text{poly}_1(n) + \text{poly}_2(n)))$ است که چندجمله‌ای می‌باشد. پس $L_1 \cdot L_2 \in \mathcal{P}$.

\paragraph{کلاس $\mathcal{NP}$:}
ماشین غیرقطعی $N$ روی ورودی $x$ به طول $n$:
\begin{enumerate}
    \item به صورت غیرقطعی یک نقطه شکست $i \in \{0, \dots, n\}$ انتخاب کن تا $x = yz$ شود.
    \item به صورت غیرقطعی $N_1(y)$ و $N_2(z)$ را اجرا کن.
    \item اگر هر دو پذیرفتند، \textbf{بپذیر}.
\end{enumerate}
حدس زدن $i$ و اجرای غیرقطعی هر دو ماشین در زمان چندجمله‌ای انجام می‌شود، پس $L_1 \cdot L_2 \in \mathcal{NP}$.

\subsection*{(د) بسته بودن تحت بستار کلینی ($L^*$)}

\paragraph{کلاس $\mathcal{P}$:}
از روش برنامه‌نویسی پویا (Dynamic Programming) استفاده می‌کنیم. برای رشته $x$ به طول $n$:
آرایه جدول $T[0..n]$ را تعریف می‌کنیم که $T[i] = \text{True}$ است اگر و تنها اگر پیشوند $x[1..i] \in L^*$ باشد.
\begin{enumerate}
    \item مقدار اولیه $T[0] = \text{True}$ قرار بده (چون $\epsilon \in L^*$).
    \item برای $i = 1$ تا $n$:
    \begin{itemize}
        \item قرار بده $T[i] = \text{False}$.
        \item برای $j = 0$ تا $i-1$:
        \begin{itemize}
            \item اگر $T[j] == \text{True}$ و $M(x[j+1..i])$ پذیرفته شد، قرار بده $T[i] = \text{True}$ و از حلقه داخلی خارج شو.
        \end{itemize}
    \end{itemize}
    \item مقدار $T[n]$ را خروجی بده.
\end{enumerate}
پیچیدگی زمانی برابر $O(n^2 \cdot \text{poly}(n))$ است که چندجمله‌ای می‌باشد. پس $L^* \in \mathcal{P}$.

\paragraph{کلاس $\mathcal{NP}$:}
ماشین غیرقطعی $N'$ روی ورودی $x$ به طول $n$:
\begin{enumerate}
    \item اگر $x = \epsilon$ باشد، \textbf{بپذیر}.
    \item به صورت غیرقطعی یک افراز برای $x$ به صورت $x = u_1 u_2 \dots u_k$ حدس بزن ($k \le n$).
    \item به صورت غیرقطعی $N(u_m)$ را برای تمام $m \in \{1, \dots, k\}$ اجرا کن.
    \item اگر تمامی $N(u_m)$‌ها ورودی را پذیرفتند، \textbf{بپذیر}.
\end{enumerate}
حدس زدن افراز و اجرای غیرقطعی روی بخش‌ها در زمان چندجمله‌ای صورت می‌گیرد. پس $L^* \in \mathcal{NP}$.
